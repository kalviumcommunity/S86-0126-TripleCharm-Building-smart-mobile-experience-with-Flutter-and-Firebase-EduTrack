rules_version = '2';\nservice cloud.firestore {\n  match /databases/{database}/documents {\n    // Helper function to check authentication\n    function isAuthenticated() {\n      return request.auth.uid != null;\n    }\n\n    // Teachers collection - Users own profile \n    match /teachers/{teacherId} {\n      // Allow user to read their own profile\n      allow read: if isAuthenticated() && request.auth.uid == teacherId;\n      // Allow account creation during signup\n      allow create: if isAuthenticated() && request.auth.uid == teacherId;\n      // Allow user to update their own profile\n      allow update: if isAuthenticated() && request.auth.uid == teacherId;\n      // Do not allow deletion\n      allow delete: if false;\n    }\n\n    // Classes collection - Filter by teacherId\n    match /classes/{classId} {\n      // User can read only their own classes\n      allow read: if isAuthenticated() \n                   && resource.data.teacherId == request.auth.uid;\n      // User can create class if teacherId matches auth user\n      allow create: if isAuthenticated() \n                     && request.resource.data.teacherId == request.auth.uid\n                     && request.resource.data.className is string\n                     && request.resource.data.description is string;\n      // User can update their own class\n      allow update: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n      // User can delete their own class\n      allow delete: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n    }\n\n    // Students collection - Filter by teacherId\n    match /students/{studentId} {\n      // User can read only their own students\n      allow read: if isAuthenticated() \n                   && resource.data.teacherId == request.auth.uid;\n      // User can create students\n      allow create: if isAuthenticated() \n                     && request.resource.data.teacherId == request.auth.uid\n                     && request.resource.data.classId is string\n                     && request.resource.data.name is string;\n      // User can update their own students\n      allow update: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n      // User can delete their own students\n      allow delete: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n    }\n\n    // Attendance collection - Filter by teacherId\n    match /attendance/{attendanceId} {\n      // User can read only their own attendance records\n      allow read: if isAuthenticated() \n                   && resource.data.teacherId == request.auth.uid;\n      // User can create attendance records\n      allow create: if isAuthenticated() \n                     && request.resource.data.teacherId == request.auth.uid\n                     && request.resource.data.studentId is string\n                     && request.resource.data.classId is string;\n      // User can update their own attendance records\n      allow update: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n      // Do not allow deletion of attendance records (keep history)\n      allow delete: if false;\n    }\n\n    // Exams collection - Filter by teacherId\n    match /exams/{examId} {\n      // User can read only their own exams\n      allow read: if isAuthenticated() \n                   && resource.data.teacherId == request.auth.uid;\n      // User can create exams\n      allow create: if isAuthenticated() \n                     && request.resource.data.teacherId == request.auth.uid\n                     && request.resource.data.classId is string\n                     && request.resource.data.examName is string\n                     && request.resource.data.totalMarks is number;\n      // User can update their own exams\n      allow update: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n      // User can delete their own exams\n      allow delete: if isAuthenticated() \n                     && resource.data.teacherId == request.auth.uid;\n    }\n\n    // Marks collection - User can write marks for their exams\n    match /marks/{markId} {\n      // Read: User can read marks if they teach the exam\n      allow read: if isAuthenticated() \n                   && exists(/databases/$(database)/documents/exams/$(resource.data.examId)) \n                   && get(/databases/$(database)/documents/exams/$(resource.data.examId)).data.teacherId == request.auth.uid;\n      // Create: User can add marks for their exams\n      allow create: if isAuthenticated() \n                     && exists(/databases/$(database)/documents/exams/$(request.resource.data.examId))\n                     && get(/databases/$(database)/documents/exams/$(request.resource.data.examId)).data.teacherId == request.auth.uid;\n      // Update: User can update marks for their exams\n      allow update: if isAuthenticated() \n                     && exists(/databases/$(database)/documents/exams/$(resource.data.examId))\n                     && get(/databases/$(database)/documents/exams/$(resource.data.examId)).data.teacherId == request.auth.uid;\n      // Delete: Do not allow deletion of marks\n      allow delete: if false;\n    }\n  }\n}\n
